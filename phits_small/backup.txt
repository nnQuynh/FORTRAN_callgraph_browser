      subroutine setmat(iom,ierr)
      use GGMARRAYMOD !2020ASTOM
      use moddas_material
      use moddas_tally
      implicit real*8 (a-h,o-z)
      parameter ( imfnmax=1000) ! T.Sato 2022/10/30
      include 'param.inc'
      include 'ggsparam.inc'
      include 'ggmparam.inc'
      common /kmat1o/ iom1, iom2, iom3
      common /tall00/ itnm, ital(itlmax), itals(itlmax), italm(itlmax)
      common /tall33/ itln(itlmax,2), itli(itlmax,2), itlr(itlmax,2),
     &                rtdm(itlmax,2)
      common /fmcard/ ifm, ifmi(imfnmax,3)
      common /regcm/  icmg(kvlmax)
      common /eparm/  esmax, esmin, emin(20)
      common /kmat1k/ kmatd(kvlmax), kmate(kvlmax)
      common /kmat1h/ kmatg(kvlmax)
      common /kmat1a/ mxmat, mxmat0, mxnel
      common /kmat1g/ kmat(kvlmax)
      common /kmat1ka/ kmathd(kvlmax), kmathe(kvlmax)
      common /emode/  emodem, ge1, ge2, iemode ! frtati 2022/03/11
      common /ndemax/ dnmax(20)
      common /dpnmaxcom/ dpnmax ! frtati 2021/12/17
      character hs*10,ht*10,hb*30,hc*181
      integer, allocatable :: iturn(:) ! frtati 2021/12/17
      do 30 i = 1, mxa
         do 10 j = 1, nmat
            if( nmt(j) .eq. mat(i) ) goto 20
   10    continue
            mat(i) = 0
            goto 30
   20    mat(i) = j
   30 continue
      do i = 1, mxa
         icmg(i) = mat(i)
      end do
      do 70 m = 1, nmat * kpt(3)
         if( mbi(m) .ge. 0 ) goto 70
         n = -mbi(m)
         mbi(m) = 0
   40    do 50 i = 1, nmat
   50    if( nmt(i) .eq. n ) goto 60
         goto 70
   60    n = -mbi(i)
         mbi(i) = 1
         if( n .gt. 0 ) goto 40
   70 continue
      mn = 1
      do 240 km = 1, mix
         if( km .ge. jmd(1+mn+1) ) mn = mn + 1
      do 230 m = 1, mipt
            if( kpt(m) .eq. 0 .and.
     &        ( m .ne. 3 .or. ides .ne. 0 .or. kpt(2) .eq. 0 ) )
     &         goto 230
         if( m .eq. 1 .or. m .eq. 9 ) then
               lem = km-jmd(1+mn)+1
            if( m .eq. 1 ) then
                  dsmax = das_kmate(kmate(mn)+(lem-1)*5+12)
                  izia = nint(das_kmatg(kmatg(mn)+(lem-1)*3+32))
                  if( dsmax .le. emin(2) ) goto 230
            else if( m .eq. 9 ) then
                  dsmax = das_kmate(kmate(mn)+(lem-1)*5+11)
                  izia = nint(das_kmatg(kmatg(mn)+(lem-1)*3+32))
                  if( dsmax .le. emin(1) ) goto 230
            end if
         end if
         if( m .eq. 31 .or. m .eq. 34 ) then
               lem = km-jmd(1+mn)+1
            if( m .eq. 31 ) then
                  dsmax = das_kmate(kmate(mn)+(lem-1)*5+14)
                  izia = nint(das_kmatg(kmatg(mn)+(lem-1)*3+32))
                  if ( dsmax .le. emin(15) ) goto 230
            else if( m .eq. 34 ) then
                  dsmax = das_kmate(kmate(mn)+(lem-1)*5+15)
                  izia = nint(das_kmatg(kmatg(mn)+(lem-1)*3+32))
                  if ( dsmax .le. emin(18) ) goto 230
            end if
         end if
      do 120 i = 1, mxa
  120 if( mat(i) .eq. mn .and.
     &  ( fim(m,i) .ne. 0. .or. m .eq. 3 .and.
     &    kpt(3) .eq. 0 .and. fim(2,i) .ne. 0 ) ) goto 150
         do j = 1, 2
            if( ( j .eq. 1 .and. m .eq. 9 ) .or.
     &          ( j .eq. 2 .and. m .eq. 1 ) ) then
               do k = 1, itnm
                  if( itln(k,j) .gt. 0 ) then
                     do kk = 1, itln(k,j)
                        if( mlib(itli(k,j)+1,kk)
     &                     .eq. nmt(mn) ) goto 150
                     end do
                  end if
               end do
            end if
         end do
      if( ifm .gt. 0 ) then
         do j = 1, ifm
            if( ifmi(j,1) .eq. m .and.
     &          ifmi(j,2) .eq. nmt(mn) ) goto 150
         end do
      end if
      goto 230
  150 write(ht(1:7),'(i6,1h.)')iza(km)
      l = kmm(km)
      ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
      if( m .eq. 9 ) goto 192
      if( m .eq. 31 ) goto 193
      if( m .eq. 34 ) goto 194
      if( m .gt. 3 ) goto 230
      goto (160,180,190) m
  160 if( ht(8:10) .ne. ' ' .and.
     &    index('pgue',ht(10:10)) .eq. 0 ) goto 168
      l = lxd(1,mn)
      ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
  168 if( mcal .ne. 0 ) ht(10:10)='m'
      if( mcal .ne. 0 .or. ht(10:10) .eq. 'y' ) goto 200
      if( ht(10:10) .eq. 'm' ) ht(10:10) = 'c'
      if( ht(10:10) .eq. ' ' ) ht(10:10) = 'c'
! T.Sato 2021/08/30. kdr is not defined. In PHITS, extention **d cannot be used
      goto 200
  180 ht(4:6) = '000'
      if( index('pg',ht(10:10)) .ne. 0 ) goto 185
      l = lxd(2,mn)
      ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
  185 if( mcal .ne. 0 ) ht(10:10) = 'g'
      if( mcal .eq. 0 ) ht(10:10) = 'p'
      goto 200
  190 ht(4:6) = '000'
      if( ht(10:10) .eq. 'e' ) goto 200
      l = lxd(3,mn)
      ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
      goto 200
  192 continue
      if( ht(10:10) .eq. 'y' ) goto 200
      if( ht(10:10) .eq. 'h' ) goto 200
      l = lxd(9,mn)
      ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
  193 if( m .eq. 31 .and. mcal .eq. 0 ) then
        l = lxd(31,mn)
        ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
      end if
  194 if( m .eq. 34 .and. mcal .eq. 0 ) then
        l = lxd(34,mn)
        ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
      end if
  200 do 210 i = 1, mxe
         call zaid(2,hs,ixl(1,i))
  210 if( hs .eq. ht ) goto 220
         mxe = mxe + 1
      if( mxe .gt. mxe1 ) then
         write(iom,'(''Error: material (mex) overflow in setmat 1'')')
         ierr = 1
         return
      end if
      call zaid(1,ht,ixl(1,i))
  220 lme(m,km) = i
  230 continue
  240 continue
      if( ispn .eq. 0 ) goto 248
         mn = 1
      do 247 km = 1, mix
         if( km .ge. jmd(1+mn+1) ) mn = mn + 1
         if( izn(km) .eq. 0 ) goto 247
         do ii = 1, mxa
           if( mat(ii).eq.mn ) goto 241
         end do
         goto 247
  241    continue
         write(ht(1:7),'(i6,1h.)') izn(km)
         l = kmm(km)
         ht(8:10) = char(mod(l,256))//
     &              char(mod(l/256,256))//char(l/65536)
         if( index('u',ht(10:10)) .ne. 0 ) goto 242
      l = lxd(4,mn)
      ht(8:10) = char(mod(l,256))//char(mod(l/256,256))//char(l/65536)
  242 do 244 i = 1, mxe
      call zaid(2,hs,ixl(1,i))
  244 if( hs .eq. ht ) goto 246
      mxe = mxe + 1
      i = mxe
      if(mxe.gt.mxe1) then
         write(iom,'(''Error: material (mex) overflow in setmat 2'')')
         ierr = 1
         return
      end if
      call zaid(1,ht,ixl(1,i))
  246 lmn(km) = i
  247 continue
  248 if( mcal .ne. 0 ) goto 262
      do 260 km = 1, indt
         call zaid(2,ht,kmt(1,km))
         ht(10:10) = 't'
         do 250 i = 1, mxe
            call zaid(2,hs,ixl(1,i))
            if( hs .eq. ht ) goto 260
            if( hs(1:6) .ne. ht(1:6) ) goto 250
            if( hs(8:9) .ne. ' ' .and. ht(8:9) .ne. ' ' ) goto 250
            write(iom1,*)
     &      '## warning. '//ht//' and '//hs//
     &      ' are both present on mt cards.'
            if( hs(8:9) .eq. ' ' ) call zaid(1,ht,ixl(1,i))
            go to 260
  250    continue
         mxe = mxe + 1
         if( mxe .gt. mxe1 ) then
            write(iom,
     &           '(''Error: material (mex) overflow in setmat 3'')')
            ierr = 1
            return
         end if
         call zaid(1,ht,ixl(1,mxe))
  260 continue
  262 if( mxe .eq. 0 ) write(iom1,*)
     &         '## warning. '//
     &         'no cross-section tables are called for in this problem.'
      if( mxe .eq. 0 ) goto 325
      do 310 ie = 1, mxe
         do 300 je = ie + 1, mxe
            call zaid(2,hs,ixl(1,ie))
            mt = index(htn,hs(10:10))
            call zaid(2,ht,ixl(1,je))
            nt = index(htn,ht(10:10))
            if( mt .lt. nt ) goto 300
            if( mt .gt. nt ) goto 270
            if( hs .le. ht ) goto 300
  270       do 280 i = 1, 3
               l = ixl(i,ie)
               ixl(i,ie) = ixl(i,je)
  280          ixl(i,je) = l
            do 290 i = 1, mipt
            do 290 j = 1, mix
               l = lme(i,j)
               if( l .eq. ie ) lme(i,j) = je
  290          if( l .eq. je ) lme(i,j) = ie
            if( ispn .eq. 0 ) goto 300
            do 295 i = 1, mix
                l = lmn(i)
                if( l .eq. ie ) lmn(i) = je
  295           if( l .eq. je ) lmn(i) = ie
  300    continue
  310 continue
      do 320 i = 1, mxe - 1
         call zaid(2,hs,ixl(1,i))
         call zaid(2,ht,ixl(1,i+1))
  320    if( hs(1:6) .eq. ht(1:6) .and. hs(10:10) .eq. ht(10:10) )
     &   write(iom1,*)
     &   '## warning. '//hs//' and '//ht//' are both called for.'
  325 call jxsdir(i,iom,ierr)
      if( ierr .ne. 0 ) return
      if( i .eq. 1 ) return
      i1_flag = 0
      allocate(iturn(1:mxe))
      mxe_reduce = 0
      do i = 1, mxe
        if( nty(i).eq.0 ) then
          mxe_reduce = mxe_reduce + 1
          iturn(i) = 0
          i1_flag = 1
          cycle
        else if( nty(i).eq.-10 ) then
          i1_flag = 1
        end if
        iturn(i) = i - mxe_reduce
      end do
      if ( i1_flag.eq.1 ) then ! set data max
      i2_flag = 0
      mn = 1
      do km = 1, mix
        if ( km .ge. jmd(1+mn+1) ) mn = mn + 1
        lem_tem = km-jmd(1+mn)+1
        if( lme(1,km).ne.0 ) then ! frtati 2022/06/13
          if( dnmax(2).gt.20.d0 ) then ! frtati 2022/03/23
            if( nty(lme(1,km)).eq.-10 ) then ! use JENDL up to 20MeV
              i2_flag = 1
              lme(1,km) = iturn(lme(1,km))
              das_kmate(kmate(mn)+(lem_tem-1)*5+12) = 20.d0
              ixl(3,lme(1,km)) = ixc(3,lme(1,km)) ! replace extension to neutron default
            else if( nty(lme(1,km)).eq.0 ) then
              i2_flag = 1
              lme(1,km) = 0
              das_kmate(kmate(mn)+(lem_tem-1)*5+12) = 0.d0
            else
              lme(1,km) = iturn(lme(1,km))
            end if
          else ! frtati 2022/06/13
            lme(1,km) = iturn(lme(1,km))
          end if
        end if
      end do
      do km = 1, mix ! frtati 2022/03/22
        if( nty(lme(1,km)).eq.-10 ) nty(lme(1,km)) = 1
      end do
      if( i2_flag.eq.1 ) then
        do m = 1, mxmat
          t1_dmax = 0.d0
          t2_dmax = 1.d10
          nel = nint( dnel_das(kmat0+m) )
          do l = 1, nel
            iz = nint( zz_das(kmat(m)+l) )
            ia = nint( a_das(kmat(m)+l) )
            mn = 1
            do km = 1, mix
              if( km .ge. jmd(1+mn+1) ) mn = mn + 1
              lem_tem = km-jmd(1+mn)+1
              itz = iza(km) / 1000
              ita = iza(km) - 1000 * itz
              if( m.eq.mn .and. iza(km).eq.1001 ) then ! frtati 2022/03/11
                das_kmatd(kmatd(m)+7) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+12)
              end if
              if( (m.eq.mn .and. iz.eq.itz) .and.
     &            (ia.eq.ita .or. ita.eq.0) ) then
                das_kmatd(kmatd(m)+(l-1)*5+12) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+12)
                exit
              end if
            end do
            if ( das_kmatd(kmatd(mn)+(l-1)*5+12).gt.t1_dmax ) then
              t1_dmax = das_kmatd(kmatd(m)+(l-1)*5+12)
            end if
            if ( das_kmatd(kmatd(mn)+(l-1)*5+12).lt.t2_dmax ) then
              t2_dmax = das_kmatd(kmatd(m)+(l-1)*5+12)
            end if
          end do
          das_kmatd(kmatd(m)+3) = t1_dmax
          das_kmatd(kmatd(m)+4) = t2_dmax
        end do
      end if
      i2_flag = 0
      do m = 1, mipt
        if ( m.eq.2 .or. m.eq.3 .or.
     &       m.eq.9 .or. m.eq.31.or. m.eq.34 ) then
          mn = 1
          do km = 1, mix
            if ( km.ge.jmd(1+mn+1) ) mn = mn + 1
            lem_tem = km-jmd(1+mn)+1
            if( lme(m,km).ne.0 ) then ! frtati 2022/03/23
              if( nty(lme(m,km)).eq.0 ) then
                i2_flag = 1
                lme(m,km) = 0
                if ( m.eq.9 ) then
                  das_kmate(kmate(mn)+(lem_tem-1)*5+11) = 0.d0
                else if ( m.eq.31 ) then
                  das_kmate(kmate(mn)+(lem_tem-1)*5+14) = 0.d0
                else if ( m.eq.34 ) then
                  das_kmate(kmate(mn)+(lem_tem-1)*5+15) = 0.d0
                end if
              else
                lme(m,km) = iturn(lme(m,km))
              end if
            end if
          end do
        end if
      end do
      if( i2_flag.eq.1 ) then
        do m = 1, mxmat
          t1_dmax = 0.d0
          t2_dmax = 1.d10
          t3_dmax = 0.d0
          t4_dmax = 1.d10
          t5_dmax = 0.d0
          t6_dmax = 1.d10
          nel = nint( dnel_das(kmat0+m) )
          do l = 1, nel
            iz = nint( zz_das(kmat(m)+l) )
            ia = nint( a_das(kmat(m)+l) )
            mn = 1
            do km = 1, mix
              if( km .ge. jmd(1+mn+1) ) mn = mn + 1
              lem_tem = km-jmd(1+mn)+1
              itz = iza(km) / 1000
              ita = iza(km) - 1000 * itz
              if( m.eq.mn .and. iza(km).eq.1001 ) then ! frtati 2022/03/11
                das_kmatd(kmatd(m)+6) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+11)
                das_kmatd(kmatd(m)+9) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+14)
                das_kmatd(kmatd(m)+10) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+15)
              end if
              if( (m.eq.mn .and. iz.eq.itz) .and.
     &            (ia.eq.ita .or. ita.eq.0) ) then
                das_kmatd(kmatd(m)+(l-1)*5+11) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+11)
                das_kmatd(kmatd(m)+(l-1)*5+14) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+14)
                das_kmatd(kmatd(m)+(l-1)*5+15) =
     &          das_kmate(kmate(m)+(lem_tem-1)*5+15)
                exit
              end if
            end do
            if ( das_kmatd(kmatd(m)+(l-1)*5+11).gt.t1_dmax ) then
              t1_dmax = das_kmatd(kmatd(m)+(l-1)*5+11)
            end if
            if ( das_kmatd(kmatd(m)+(l-1)*5+11).lt.t2_dmax ) then
              t2_dmax = das_kmatd(kmatd(m)+(l-1)*5+11)
            end if
            if ( das_kmatd(kmatd(m)+(l-1)*5+14).gt.t3_dmax ) then
              t3_dmax = das_kmatd(kmatd(m)+(l-1)*5+14)
            end if
            if ( das_kmatd(kmatd(m)+(l-1)*5+14).lt.t4_dmax ) then
              t4_dmax = das_kmatd(kmatd(m)+(l-1)*5+14)
            end if
            if ( das_kmatd(kmatd(m)+(l-1)*5+15).gt.t5_dmax ) then
              t5_dmax = das_kmatd(kmatd(m)+(l-1)*5+15)
            end if
            if ( das_kmatd(kmatd(m)+(l-1)*5+15).lt.t6_dmax ) then
              t6_dmax = das_kmatd(kmatd(m)+(l-1)*5+15)
            end if
          end do
          das_kmatd(kmatd(m)+1) = t1_dmax
          das_kmatd(kmatd(m)+2) = t2_dmax
          das_kmathd(kmathd(m)+3) = t3_dmax
          das_kmathd(kmathd(m)+4) = t4_dmax
          das_kmathd(kmathd(m)+5) = t5_dmax
          das_kmathd(kmathd(m)+6) = t6_dmax
        end do
      end if
      if( ispn.eq.1 ) then
        i2_flag = 0
        mn = 1
        do km = 1, mix
          if ( km .ge. jmd(1+mn+1) ) mn = mn + 1
          lem_tem = km-jmd(1+mn)+1
          if( lmn(km).ne.0 ) then ! frtati 2022/03/23
            if( nty(lmn(km)).eq.0 ) then
              i2_flag = 1
              lmn(km) = 0
              das_kmate(kmate(mn)+(lem_tem-1)*5+13) = 0.d0
            else
              lmn(km) = iturn(lmn(km))
            end if
          end if
        end do
        if( i2_flag.eq.1 ) then
          do m = 1, mxmat
            t1_dmax = 0.d0
            t2_dmax = 1.d10
            nel = nint( dnel_das(kmat0+m) )
            do l = 1, nel
              iz = nint( zz_das(kmat(m)+l) )
              ia = nint( a_das(kmat(m)+l) )
              mn = 1
              do km = 1, mix
                if( km .ge. jmd(1+mn+1) ) mn = mn + 1
                lem_tem = km-jmd(1+mn)+1
                itz = iza(km) / 1000
                ita = iza(km) - 1000 * itz
                if( (m.eq.mn .and. iz.eq.itz) .and.
     &              (ia.eq.ita .or. ita.eq.0) ) then
                  das_kmatd(kmatd(m)+(l-1)*5+13) =
     &            das_kmate(kmate(m)+(lem_tem-1)*5+13)
                  exit
                end if
              end do
              if ( das_kmatd(kmatd(m)+(l-1)*5+13).gt.t1_dmax ) then
                t1_dmax = das_kmatd(kmatd(m)+(l-1)*5+13)
              end if
              if ( das_kmatd(kmatd(m)+(l-1)*5+13).lt.t2_dmax ) then
                t2_dmax = das_kmatd(kmatd(m)+(l-1)*5+13)
              end if
            end do
            das_kmathd(kmathd(m)+1) = t1_dmax
            das_kmathd(kmathd(m)+2) = t2_dmax
          end do
        end if
      end if
      do i = 1, mxe
        if ( iturn(i).ne.0 ) then
          nty(iturn(i)) = nty(i)
          ixl(:,iturn(i)) = ixl(:,i)
          ixc(:,iturn(i)) = ixc(:,i)
        end if
      end do
      mxe = mxe - mxe_reduce
      end if
      deallocate(iturn)
      if( iemode.ne.0 ) then
        do iex = 1, mxe
          mn = 1
          do km = 1, mix
            if( km .ge. jmd(1+mn+1) ) mn = mn + 1
            lem_tem = km-jmd(1+mn)+1
            if( iza(km).eq.1001 .and.
     &         (lme(1,km).eq.iex .or. lme(9,km).eq.iex .or.
     &          lme(31,km).eq.iex .or. lme(34,km).eq.iex) ) then
              if( lme(1,km).eq.iex ) then
                das_kmatd(kmatd(mn)+7) =
     &          min(das_kmate(kmate(mn)+(lem_tem-1)*5+12),20.d0)
                das_kmate(kmate(mn)+(lem_tem-1)*5+12) =
     &          min(das_kmate(kmate(mn)+(lem_tem-1)*5+12),20.d0)
              else if( lme(9,km).eq.iex ) then
                das_kmatd(kmatd(mn)+6) = 0.d0
                das_kmate(kmate(mn)+(lem_tem-1)*5+11) = 0.d0
              else if( lme(31,km).eq.iex ) then
                das_kmatd(kmatd(mn)+9) = 0.d0
                das_kmate(kmate(mn)+(lem_tem-1)*5+14) = 0.d0
              else if( lme(34,km).eq.iex ) then
                das_kmatd(kmatd(mn)+10) = 0.d0
                das_kmate(kmate(mn)+(lem_tem-1)*5+15) = 0.d0
              end if
            end if
          end do
        end do
      end if
      do 350 i = 1, mxe
         hc=' '
         call zaid(2,hc,ixc(1,i))
         do 330 j = 11, 181
  330    hc(j:j) = char(mod(ixc((j+1)/3,i)/256**mod(j+1,3),256))
         call nxtsym(hc,' ',11,it,iu,1)
         if( iu .eq. 0 ) goto 350
         hb = hc(it:iu)

      subroutine eldn1(az,su,pl,fo,eq,xq,nt,wt,tm)
      use GGMARRAYMOD !2020ASTOM
      implicit real*8 (a-h,o-z)
      include 'param.inc'
      include 'ggsparam.inc'
      include 'ggmparam.inc'
      common /kmat1o/ iom1, iom2, iom3
      dimension fo(*),eq(*),xq(*)
         nt = 0
      do 20 j = jmd(1+mkc), jmd(1+mkc+1) - 1
         iz = iza(j) / 1000
         ns = nxs(11,lme(3,j))
      do 20 n = 1, ns
         zs = exs(jxs(11,lme(3,j))+n-1)
         nt = nt + 1
         fo(nt) = fme(j) * abs(zs) / az
         if( jcond(mkc) .ne. 0 .and. zs .le. 0. ) goto 10
         xq(nt) = (exs(jxs(11,lme(3,j))+ns+n-1)/pl)**2
         eq(nt) = 2. * third * fo(nt)
         goto 20
   10    xq(nt) = 0.
         eq(nt) = fo(nt)
   20 continue
         mt = 0
         wt = 2.
   30    xo = wt
         fu = 0.
         dr = 0.
      do 40 n = 1, nt
         pa = wt * wt * xq(n) + eq(n)
         fu = fu + fo(n) * log(pa)
   40    dr = dr + fo(n) * 2. * wt * xq(n) / pa
         wt = wt - (fu-su) / dr
         mt = mt + 1
         if( mt .lt. 25 ) goto 50
         write(iom1,*)
     &   '## warning. electron oscillator fitting convergence problem.'
         wt = 1.e-4
         goto 70
   50    if( wt .gt. 0. ) goto 60
         wt = .5 * xo
         goto 30
   60    if( abs(wt-xo) .gt. 1.e-6 * xo ) goto 30
   70    w2 = wt**2
      do 80 l = 1, nt
         xq(l) = w2 * xq(l)
   80    eq(l) = xq(l) + eq(l)
         tm = 0.
         if( jcond(mkc) .gt. 0 ) return
         sm = 0.
      do 90 l = 1, nt
   90    sm = sm + fo(l) / xq(l)
         bq = one / (one+sm)
         tm = one / sqrt(one-bq) - one
      return
      end
      subroutine eldn2(s,tm,fo,eq,xq,nt,wt,d)
      use GGMARRAYMOD !2020ASTOM
      implicit real*8 (a-h,o-z)
      include 'param.inc'
      include 'ggsparam.inc'
      include 'ggmparam.inc'
      common /kmat1o/ iom1, iom2, iom3
      dimension fo(*),eq(*),xq(*)
         d = 0.
         if( s .le. tm ) return
         mt = 0
         x = s * (s+2.)
         re = 1. / x
   10    xo = x
         fn = 0.
         dv = 0.
      do 20 n = 1, nt
         fn = fn + fo(n) / (xq(n)+x)
   20    dv = dv - fo(n) / ((xq(n)+x)**2)
         x = x - (fn-re) / dv
         mt = mt + 1
         if( mt .lt. 25 ) goto 30
       write(iom1,*)
     & '## warning. unconverged density effect correction set to zero.'
         d = 0.
         return
   30    if( x .ge. 0. ) goto 40
         x = .5 * xo
         goto 10
   40    if( abs(x-xo) .gt. 1.e-6 * xo ) goto 10
   50    d = -x / (s+1.)**2
      do 60 l = 1, nt
   60    d = d + fo(l) * log(one+x/eq(l))
      return
      end
      subroutine elscat(az,iom,ierr)
      use GGMARRAYMOD !2020ASTOM
      implicit real*8 (a-h,o-z)

      include 'ggmparam.inc'
      dimension p(240),h(240)

         z = iza(j) / 1000
         cr = cr + z**2 * fme(j)
   10    cd = cd + (z+1.) * z * fme(j)

      do 110 n = 1, nee - 1

         e2 = (eee(n)-eee(n+1)) / nsb(mkc)

         r2 = drs(n+nee*(mkc-1))

   20    p(i) = 0.

         call elsng(2,.5*(e1-e2),.5*(r1-r2),p,lx,iom,ierr)
         if( ierr .ne. 0 ) return
         ta = p(1) * (r1-r2) / (p(2)*(r1+r2))
         xf = 2. * r2 * p(1) / (p(2)*(r1+r2)-p(1)*(r1-r2))
         hm = ta**(xf*p(lx))

      function xsecmc(s,dk,wt,eo,po,es,z,ze,zf)
      use GGMARRAYMOD !2020ASTOM
      implicit real*8 (a-h,o-z)
      include 'param.inc'
      include 'ggsparam.inc'
      include 'ggmparam.inc'
      parameter (pf=2.*pie/fscon)
      dimension zf(3)
         e = eo - dk
         p = sqrt((s-dk+2.)*(s-dk))
         ey = log(e+p)
         ee = eo * e
         pp = po * p
         f1 = 2. * third * pp**3 - ee * (p**2+po**2) * pp
     &      + es * e * p**3 + ey * (eo*po**3-2.*es*pp**2)
         f2 = 8. * third * ee * pp**3 + dk**2 * (ee+ee**2+pp**2) * pp
     &      + dk * ((ee+po**2)*es*p**3-(ee+p**2)*ey*po**3)
         s4 = .5 * p * (f1*pp+log((ee+pp-1.)/dk)*f2) / pp**4
         g = 100. * dk / (ee*zf(1))
         h1 = 1. + (e/eo)**2
         h2 = 2. * third * e / eo
         if( g .gt. .735 ) p1 = (19.83-4.184*log(g+.7))*(h1-h2)
         if( g .le. .735 ) p1 = (20.867-g*(4.409-1.156*g))*h1
     &                        - (20.209-g*(2.625+ .159*g))*h2
         xsecmc = ( s4 + po * (.25*p1-(zf(2)*third+wt*zf(3)
     &                      + log(2.*ee/dk)-.5)
     &                      * (h1-h2))) * ze * e
     &                      / ((1.-exp(-pf*z*e/p))*p*eo)
      return
      end

      do 20 k = 1, nwng
   20    ek(k) = rka(k) * s
         eo = s + 1.
         eb = eo**2
         p2 = s * (s+2.)
         po = sqrt(p2)
         b = po / eo
         d = eo - po
         s2 = 4. * (2.*eb+1.) / (p2*d**3)
      do 30 l = 1, mpng
         xi(l) = (1.+b*(n1-l)/r2)/(1.+b)
   30    ch(l) = (n1-l+r2*b)/(r2+b*(n1-l))
      do 150 k = 1, nwng
         e = eo - ek(k)
         ee = eo * e
         f2 = eb + e**2
         if( s .gt. 4. ) goto 50
         p = sqrt((s-ek(k)+2.) * (s-ek(k)))
         pp = po * p
         e2 = ek(k)**2
         ef = p2 - ek(k) * eo
         pf = sqrt(p2*(p2-2.*ek(k)*eo+e2))
         if(ef.le.pf) el = log(b**2*(ef+pf)/(.5*e2*(1.-b**2)))/pp
         if(ef.gt.pf) el = log((ef+pf)/(ef-pf))/pp
         ey = log((e+p)/(e-p))
         f = ca*p/po
         f3 = (5.*eb+2.*ee+3.)/(d*p2)
         f4 = (p2-e2)/d
         f5 = 2.*(e/p2-ey/p)
         f6 = 2.*eo*(3.*ek(k)-p2*e)/(p2*d**3)
         f7 = (2.*eb*f2+1.-(6.*eb-3.*ee+f2))/(d*p2)
         f8 = ek(k)*(eb+ee-1.)/p2
      do 40 l = 1, mpng
         q2 = p2 + e2 - 2. * po * ek(k) * ch(l)
         q = sqrt(q2)
   40    y(l) = f * (s2*(xi(l)**3*(1.-ch(l)**2))-(f3+f4/q2)*xi(l)
     &        + f5 + el*(f6*xi(l)**3*(1.-ch(l)**2)+f7*xi(l)+f8)
     &        + (log((q+p)/(q-p))/(p*q))*(2.*xi(l)/d
     &        - 3.*ek(k)-ek(k)*(p2-e2)/q2))/xi(l)
         goto 130
   50 do 60 l = 1, mpng
   60    y(l) = 0.
         if( s .gt. 30. ) goto 110
      do 100 j = jmd(1+mkc), jmd(1+mkc+1) - 1
         z = iza(j)/1000
         if(eb.gt.ek(k)*(eo+20./(3.*z**third))) goto 80
         p = sqrt((s-ek(k)+2.)*(s-ek(k)))
         pp = po*p
         e2 = ek(k)**2
         ef = p2-ek(k)*eo
         pf = sqrt(p2*(p2-2.*ek(k)*eo+e2))
         if(ef.le.pf) el = log(b**2*(ef+pf)/(.5*e2*(1.-b**2)))/pp
         if(ef.gt.pf) el = log((ef+pf)/(ef-pf))/pp
         ey = log((e+p)/(e-p))
         f = (z+1.)*z*fme(j)*p/po
         f3 = (5.*eb+2.*ee+3.)/(d*p2)
         f4 = (p2-e2)/d
         f5 = 2.*(e/p2-ey/p)
         f6 = 2.*eo*(3.*ek(k)-p2*e)/(p2*d**3)
         f7 = (2.*eb*f2+1.-(6.*eb-3.*ee+f2))/(d*p2)
         f8 = ek(k)*(eb+ee-1.)/p2
      do 70 l = 1, mpng
         q2 = p2+e2-2.*po*ek(k)*ch(l)
         q = sqrt(q2)
   70    y(l) = y(l)+f*(s2*(xi(l)**3*(1.-ch(l)**2))-(f3+f4/q2)*xi(l)
     &        + f5 + el*(f6*xi(l)**3*(1.-ch(l)**2)+f7*xi(l)+f8)
     &        + (log((q+p)/(q-p))/(p*q))*(2.*xi(l)/d
     &        - 3.*ek(k)-ek(k)*(p2-e2)/q2))/xi(l)
         goto 100
   80    f = 4.*(z+1.)*z*fme(j)*b/eo
         g = log(ek(k)/(2.*ee))+2.
         a = 12.*z**third*ee/(121.*ek(k))
      do 90 l = 1, mpng
         sf = screen(a*xi(l))-g
   90    y(l) = y(l)+f*(f2*(3.+2.*sf)-2.*ee*(1.+4.*(1.-xi(l))*xi(l)*sf))
  100 continue
         goto 130
  110 do 120 j = jmd(1+mkc), jmd(1+mkc+1) - 1
         ix = lme(3,j)
         z = nxs(2,ix)
         f = 4.*(z+1.)*z*fme(j)*b/eo
         g = log(ek(k)/(2.*ee))+2.+exs(jxs(8,ix)+2)
         a = 12.*exs(jxs(8,ix))*ee/(121.*ek(k))
      do 120 l = 1, mpng
         sf = screen(a*xi(l))-g
  120    y(l) = y(l)+f*(f2*(3.+2.*sf)-2.*ee*(1.+4.*(1.-xi(l))*xi(l)*sf))
  130    ac((mpng+1)/2) = 0.
      do 140 l = 3, mpng, 2
         m = (mpng+2-l)/2
  140    ac(m) = ac(m+1)+y(l-2)+4.*y(l-1)+y(l)
      do 150 l = 1, mpng
  150    ech(l,k,n/nstp+1+(nee/nstp+1)*(mkc-1)) = min(max(zero,
     &        qpol((mpng-l)*(ac(1)/(mpng-1)),ac,cm,(mpng+1)/2)),one)
      return
      end
      subroutine tmpneu
      use GGMARRAYMOD !2020ASTOM
      implicit real*8 (a-h,o-z)
      include 'param.inc'
      include 'ggsparam.inc'
      include 'ggmparam.inc'
         k = 0
         th = 0.
      do 40 ic = 1, mxa
         j = mat(ic)
      do 10 m = jmd(1+j), jmd(1+j+1) - 1
   10    if(lme(1,m).eq.iex) goto 20
         goto 40
   20    if(k.eq.0) t = tmp((ic-1)*mxt+1)
         k = 1
         er = .01
         if(dbcn(20).ne.0.) er = 0.
      do 30 l = 1, mxt
   30    if(abs(tmp((ic-1)*mxt+1)-t).gt.er*t) goto 50
   40    continue
         if(k.eq.0) return
         th = t
   50    if(abs(tbt(iex)-th).le.er*th) return
         b = 500.*abs(th-tbt(iex))/awn(iex)
      do 80 j = 1, nxs(3,iex)
         l = jxs(1,iex)+j-1
         if(xss(l).gt.b) goto 90
         f1 = 1.
         if(tbt(iex).eq.0.) goto 60
         a = sqrt(awn(iex)*xss(l)/tbt(iex))
         if(a.ge.2.) f1 = 1.+.5/a**2
         i = a/.04
         if(a.lt.2.) f1 = (thgf(i)+(a/.04-i)*(thgf(i+1)-thgf(i)))/a
   60    f2 = 1.
         if(th.eq.0.) goto 70
         a = sqrt(awn(iex)*xss(l)/th)
         if(a.ge.2.) f2 = 1.+.5/a**2
         i = a/.04
         if(a.lt.2.) f2 = (thgf(i)+(a/.04-i)*(thgf(i+1)-thgf(i)))/a
   70    a = xss(l+3*nxs(3,iex))*(f2-f1)/f1
         xss(l+nxs(3,iex)) = xss(l+nxs(3,iex))+a
   80    xss(l+3*nxs(3,iex)) = xss(l+3*nxs(3,iex))+a
   90    tbt(iex) = th
      return
      end
